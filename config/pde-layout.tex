% Project layout: wider main text, narrower notes, and full-width part pages.
% With margin=right in main.tex, all margin notes stay on the right.
% Explicit A4 avoids geometry's unrelated, global "screen" paper-size option.
\geometry{a4paper,top=24mm,bottom=24mm,left=22mm,right=56mm,
  marginparwidth=38mm,marginparsep=6mm,bindingoffset=0mm}
\normalmarginpar

% Folios are centered on the physical page, including the annotation column.
% A page anchor keeps the position stable across normal and full-width pages.
\newcommand{\PDECenteredFolio}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=base,inner sep=0pt,text=LumenMuted,
      font=\normalfont\sffamily\small]
      at ([yshift=12mm]current page.south) {\thepage};
  \end{tikzpicture}}
\fancyfoot{}
\fancyfoot[C]{\PDECenteredFolio}
\fancypagestyle{plain}{%
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \renewcommand{\footrulewidth}{0pt}
  \fancyfoot[C]{\PDECenteredFolio}}

% Contents use the full page width, with clear part/chapter/section hierarchy.
% The section sign is decorative: source numbering and bookmarks stay intact.
\contentsmargin{2.8em}
\titlecontents{part}[0pt]
  {\addvspace{17pt}\large\sffamily\bfseries\color{LumenAccent}}
  {\thecontentslabel\enspace}{}
  {\hfill\normalsize\contentspage}
  [\addvspace{4pt}]
\titlecontents{chapter}[2.4em]
  {\addvspace{10pt}\normalsize\bfseries\color{LumenInk}}
  {\contentslabel[\textcolor{LumenAccent}{\thecontentslabel}]{2.4em}}{}
  {\hfill\textcolor{LumenAccent}{\contentspage}}
\titlecontents{section}[5.3em]
  {\addvspace{2pt}\small\color{LumenMuted}}
  {\contentslabel[\textcolor{LumenAccent}{\S\enspace\thecontentslabel}]{3.8em}}{}
  {\titlerule*[.65em]{\textcolor{LumenRule}{.}}%
    \textcolor{LumenAccent}{\contentspage}}
\NewCommandCopy{\PDEOriginalContents}{\tableofcontents}
\renewcommand{\tableofcontents}{%
  \newgeometry{top=24mm,bottom=24mm,left=22mm,right=22mm,
    marginparwidth=0pt,marginparsep=0pt,bindingoffset=0mm}%
  \PDEOriginalContents
  \restoregeometry}

% Keep book's standard part typography, numbering, and bookmark handling.
% Temporarily remove the annotation column so titles are centered on the page.
% Starred and optional short titles remain supported.
\NewCommandCopy{\PDEOriginalPart}{\part}
\RenewDocumentCommand{\part}{s o m}{%
  \newgeometry{top=24mm,bottom=24mm,left=22mm,right=22mm,
    marginparwidth=0pt,marginparsep=0pt,bindingoffset=0mm}%
  \IfBooleanTF{#1}{\PDEOriginalPart*{#3}}{%
    \IfNoValueTF{#2}{\PDEOriginalPart{#3}}{\PDEOriginalPart[#2]{#3}}}%
  \restoregeometry}
